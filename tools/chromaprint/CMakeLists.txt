cmake_minimum_required(VERSION 3.20)
project(synk_chromaprint LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../.tools/acoustid" ABSOLUTE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
add_subdirectory("${DEPS}/godot-cpp-godot-4.5-stable" godot-cpp)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
 set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
else()
 set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
endif()
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(FFT_LIB kissfft CACHE STRING "" FORCE)
set(KISSFFT_ROOT "${DEPS}/kissfft-131.1.0" CACHE PATH "" FORCE)
add_subdirectory("${DEPS}/chromaprint-1.6.2" chromaprint)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
 add_library(synk_chromaprint STATIC fingerprint.cpp)
else()
 add_library(synk_chromaprint SHARED fingerprint.cpp)
endif()
target_include_directories(synk_chromaprint PRIVATE "${DEPS}/chromaprint-1.6.2/src")
target_link_libraries(synk_chromaprint PRIVATE godot-cpp chromaprint)
if(WIN32)
 target_compile_options(godot-cpp PRIVATE -include cstdlib)
 target_link_options(synk_chromaprint PRIVATE -static-libgcc -static-libstdc++)
 target_link_options(chromaprint PRIVATE -static-libgcc -static-libstdc++)
endif()
